Menu.resetProgress = function() {
  var ui = SpreadsheetApp.getUi();
  var response = ui.alert('Сброс прогресса', 'Вы уверены, что хотите сбросить весь прогресс игрока?', ui.ButtonSet.YES_NO);
  
  if (response == ui.Button.YES) {
    var profileSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Player_Profile');
    profileSheet.getRange('B3').setValue(1);
    profileSheet.getRange('B4:B6').setValue(1);
    profileSheet.getRange('B7').setValue(500);
    profileSheet.getRange('B9').setValue(0);
    profileSheet.getRange('B10').setValue(100);
    profileSheet.getRange('B13').setValue(6);
    profileSheet.getRange('B14:B17').setValue(0);
    profileSheet.getRange('B19:B25').setValue(0);
    profileSheet.getRange('B26').setValue(0);
    profileSheet.getRange('B30:B31').setValue('FALSE');
    
    ui.alert('✅ Прогресс сброшен! Игрок возвращен к начальным значениям.');
  }
};

};
