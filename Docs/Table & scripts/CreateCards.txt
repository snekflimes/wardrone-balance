function createGroupedSupportCards() {
  var spreadsheet = SpreadsheetApp.getActiveSpreadsheet();
  var sheetName = 'Карточки поддержки (группировка)';
  
  // Если лист уже существует, предложим перезаписать
  var existingSheet = spreadsheet.getSheetByName(sheetName);
  if (existingSheet) {
    var ui = SpreadsheetApp.getUi();
    var response = ui.alert('Лист уже существует', 'Лист "' + sheetName + '" уже существует. Перезаписать его?', ui.ButtonSet.YES_NO);
    if (response != ui.Button.YES) {
      return;
    }
    spreadsheet.deleteSheet(existingSheet);
  }
  
  var sheet = spreadsheet.insertSheet(sheetName);
  sheet.clear();
  
  // Определение карточек (14 штук)
  var cards = [
    { id:1, name:'Рой дронов', rarity:'Редкая', type:'Призыв',
      param1:'Количество дронов', param2:'Урон дрона',
      p1_base:5, p1_growth:'linear', p1_inc:1,
      p2_base:10, p2_growth:'exponential', p2_mult:1.15 },
    { id:2, name:'Бочка с минами', rarity:'Эпическая', type:'Заклинание',
      param1:'Количество мин', param2:'Урон мины',
      p1_base:3, p1_growth:'linear', p1_inc:1,
      p2_base:50, p2_growth:'exponential', p2_mult:1.2 },
    { id:3, name:'Вертолет', rarity:'Редкая', type:'Призыв',
      param1:'Здоровье', param2:'Урон',
      p1_base:200, p1_growth:'exponential', p1_mult:1.2,
      p2_base:30, p2_growth:'exponential', p2_mult:1.15 },
    { id:4, name:'Десант БТР', rarity:'Эпическая', type:'Призыв',
      param1:'Количество БТР', param2:'Уровень БТР',
      p1_base:1, p1_growth:'step', p1_steps:[[1,4,1],[5,7,2],[8,10,3]],
      p2_base:1, p2_growth:'linear', p2_inc:1 },
    { id:5, name:'Десант танк', rarity:'Легендарная', type:'Призыв',
      param1:'Количество танков', param2:'Уровень танка',
      p1_base:1, p1_growth:'constant',
      p2_base:1, p2_growth:'linear', p2_inc:1 },
    { id:6, name:'Десант бронеавто', rarity:'Редкая', type:'Призыв',
      param1:'Количество авто', param2:'Уровень авто',
      p1_base:1, p1_growth:'step', p1_steps:[[1,2,1],[3,4,2],[5,6,3],[7,8,4],[9,10,5]],
      p2_base:1, p2_growth:'linear', p2_inc:1 },
    { id:7, name:'Десант пехота', rarity:'Обычная', type:'Призыв',
      param1:'Количество пехоты', param2:'Уровень пехоты',
      p1_base:3, p1_growth:'linear', p1_inc:1,
      p2_base:1, p2_growth:'linear', p2_inc:1 },
    { id:8, name:'Патроны - Hydra', rarity:'Обычная', type:'Ресурс',
      param1:'Количество патронов', param2:'-',
      p1_base:5, p1_growth:'linear', p1_inc:2,
      p2_base:null },
    { id:9, name:'Патроны - Hellfire', rarity:'Редкая', type:'Ресурс',
      param1:'Количество патронов', param2:'-',
      p1_base:3, p1_growth:'linear', p1_inc:1,
      p2_base:null },
    { id:10, name:'Истребитель Raptor', rarity:'Легендарная', type:'Призыв',
      param1:'Количество ракет', param2:'Урон ракеты',
      p1_base:5, p1_growth:'linear', p1_inc:1,
      p2_base:100, p2_growth:'exponential', p2_mult:1.15 },
    { id:11, name:'ЭМИ', rarity:'Эпическая', type:'Заклинание',
      param1:'Длительность (сек)', param2:'Бонус урона (%)',
      p1_base:3, p1_growth:'linear', p1_inc:0.5,
      p2_base:50, p2_growth:'linear', p2_inc:5 },
    { id:12, name:'Ремонтный дрон', rarity:'Редкая', type:'Поддержка',
      param1:'Лечение (ХП/сек)', param2:'Длительность (сек)',
      p1_base:20, p1_growth:'exponential', p1_mult:1.15,
      p2_base:5, p2_growth:'linear', p2_inc:1 },
    { id:13, name:'Отражающее поле', rarity:'Эпическая', type:'Защита',
      param1:'Длительность (сек)', param2:'Отражение (%)',
      p1_base:3, p1_growth:'linear', p1_inc:0.5,
      p2_base:50, p2_growth:'linear', p2_inc:5 },
    { id:14, name:'Кассетная бомба', rarity:'Эпическая', type:'Заклинание',
      param1:'Количество взрывов', param2:'Урон взрыва',
      p1_base:6, p1_growth:'linear', p1_inc:1,
      p2_base:30, p2_growth:'exponential', p2_mult:1.2 }
  ];
  
  // Стоимость улучшения по редкости и уровню (целевой уровень)
  var upgradeCost = {
    2: {cards:2, common:5, rare:20, epic:50, legendary:100},
    3: {cards:4, common:20, rare:50, epic:150, legendary:300},
    4: {cards:10, common:50, rare:150, epic:400, legendary:800},
    5: {cards:20, common:150, rare:400, epic:1000, legendary:2000},
    6: {cards:50, common:400, rare:1000, epic:2000, legendary:5000},
    7: {cards:100, common:1000, rare:2000, epic:4000, legendary:10000},
    8: {cards:200, common:2000, rare:4000, epic:8000, legendary:20000},
    9: {cards:400, common:4000, rare:8000, epic:20000, legendary:50000},
    10:{cards:800, common:8000, rare:20000, epic:50000, legendary:100000}
  };
  
  function getCost(rarity, targetLevel) {
    var row = upgradeCost[targetLevel];
    if (!row) return {cards:0, coins:0};
    var coins = 0;
    if (rarity === 'Обычная') coins = row.common;
    else if (rarity === 'Редкая') coins = row.rare;
    else if (rarity === 'Эпическая') coins = row.epic;
    else if (rarity === 'Легендарная') coins = row.legendary;
    return {cards: row.cards, coins: coins};
  }
  
  var currentRow = 1;
  
  for (var c = 0; c < cards.length; c++) {
    var card = cards[c];
    
    // Заголовок карточки (объединённая ячейка)
    var titleRange = sheet.getRange(currentRow, 1, 1, 5);
    titleRange.merge();
    titleRange.setValue(card.id + '. ' + card.name + ' (' + card.rarity + ', ' + card.type + ')');
    titleRange.setFontWeight('bold').setBackground('#e6b8af');
    currentRow++;
    
    // Шапка таблицы для этой карточки
    var headerValues = ['Уровень', card.param1];
    var headerCols = 2; // минимум два столбца
    if (card.param2 !== '-') {
      headerValues.push(card.param2);
      headerCols = 3;
    }
    headerValues.push('Карточек до след. ур.', 'Монет до след. ур.');
    
    var headerRange = sheet.getRange(currentRow, 1, 1, headerValues.length);
    headerRange.setValues([headerValues]);
    headerRange.setFontWeight('bold').setBackground('#d9ead3');
    currentRow++;
    
    // Данные по уровням
    for (var level = 1; level <= 10; level++) {
      var p1_val, p2_val;
      
      // Параметр 1
      if (card.p1_growth === 'linear') {
        p1_val = card.p1_base + card.p1_inc * (level - 1);
      } else if (card.p1_growth === 'exponential') {
        p1_val = card.p1_base * Math.pow(card.p1_mult, level - 1);
        p1_val = Math.round(p1_val * 100) / 100;
      } else if (card.p1_growth === 'constant') {
        p1_val = card.p1_base;
      } else if (card.p1_growth === 'step') {
        p1_val = card.p1_base;
        for (var s = 0; s < card.p1_steps.length; s++) {
          var step = card.p1_steps[s];
          if (level >= step[0] && level <= step[1]) {
            p1_val = step[2];
            break;
          }
        }
      }
      
      // Параметр 2
      if (card.p2_base === null) {
        p2_val = null;
      } else if (card.p2_growth === 'linear') {
        p2_val = card.p2_base + card.p2_inc * (level - 1);
      } else if (card.p2_growth === 'exponential') {
        p2_val = card.p2_base * Math.pow(card.p2_mult, level - 1);
        p2_val = Math.round(p2_val * 100) / 100;
      } else if (card.p2_growth === 'constant') {
        p2_val = card.p2_base;
      }
      
      var nextLevel = level + 1;
      var cost = (nextLevel <= 10) ? getCost(card.rarity, nextLevel) : {cards:0, coins:0};
      
      var rowData = [level, p1_val];
      if (card.param2 !== '-') {
        rowData.push(p2_val !== null ? p2_val : '-');
      }
      rowData.push(cost.cards, cost.coins);
      
      sheet.getRange(currentRow, 1, 1, rowData.length).setValues([rowData]);
      currentRow++;
    }
    
    // Пустая строка-разделитель
    currentRow++;
  }
  
  // Автоматическое изменение ширины столбцов
  for (var i = 1; i <= 5; i++) {
    sheet.autoResizeColumn(i);
  }
  
  SpreadsheetApp.getUi().alert('✅ Лист "' + sheetName + '" успешно создан!');
}

// Добавим пункт в меню для удобства
function onOpen() {
  var ui = SpreadsheetApp.getUi();
  ui.createMenu('🎮 War Drone')
    .addItem('Создать карточки поддержки (группировка)', 'createGroupedSupportCards')
    .addToUi();
}
