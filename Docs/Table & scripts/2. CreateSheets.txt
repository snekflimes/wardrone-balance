// Вспомогательная функция для проверки существования листа
CreateSheets.sheetExists = function(spreadsheet, sheetName) {
  var sheets = spreadsheet.getSheets();
  for (var i = 0; i < sheets.length; i++) {
    if (sheets[i].getName() === sheetName) return true;
  }
  return false;
};

CreateSheets.createWeapons = function(spreadsheet) {

  // Добавлены столбцы: H - Время перезарядки, I - Время до опустошения, J - Устойчивый DPS

  for (var level = 1; level <= maxLevel; level++) {

      'Пулемёт',
      level,

      '=F' + (level+1) + '/(D' + (level+1) + '/60)',
      '=E' + (level+1) + '*I' + (level+1) + '/(I' + (level+1) + '+H' + (level+1) + ')'
    ]);
  }
  
  data.push(['', '', '', '', '', '', '', '', '', '']);
  
  // Hydra-70
  for (var level = 1; level <= maxLevel; level++) {
    data.push([
      'Hydra-70',
      level,
      '=Constants!$B$4*POWER(Constants!$B$6,' + (level-1) + ')',
      '=Constants!$B$23*POWER(Constants!$B$24,' + (level-1) + ')',

      'Hellfire',

    1: {cards:0, common:0, rare:0, epic:0, legendary:0},
    2: {cards:2, common:5, rare:20, epic:50, legendary:100},
    3: {cards:4, common:20, rare:50, epic:150, legendary:300},
    4: {cards:10, common:50, rare:150, epic:400, legendary:800},

    9: {cards:400, common:4000, rare:8000, epic:20000, legendary:50000},
    10:{cards:800, common:8000, rare:20000, epic:50000, legendary:100000}
  };
  
  function getCost(rarity, level) {
    var row = upgradeCosts[level];
    if (!row) return {cards:0, coins:0};
    var coins = 0;
    if (rarity === 'Обычная') coins = row.common;
    else if (rarity === 'Редкая') coins = row.rare;
    else if (rarity === 'Эпическая') coins = row.epic;
    else if (rarity === 'Легендарная') coins = row.legendary;
    return {cards: row.cards, coins: coins};

      // Параметр 1
      if (card.param1_growth === 'linear') {
        p1 = card.param1_base + card.param1_inc * (level - 1);
      } else if (card.param1_growth === 'exponential') {
        p1 = Math.round(card.param1_base * Math.pow(card.param1_mult, level - 1) * 100) / 100;
      } else if (card.param1_growth === 'constant') {
        p1 = card.param1_base;
      } else if (card.param1_growth === 'step') {
        p1 = card.param1_base;
        for (var s = 0; s < card.param1_steps.length; s++) {
          var step = card.param1_steps[s];
          if (level >= step[0] && level <= step[1]) {
            p1 = step[2];
            break;
          }
        }
      }
      
      // Параметр 2
      if (card.param2_base === null) {
        p2 = null;

      } else if (card.param2_growth === 'constant') {
        p2 = card.param2_base;
      }
      
      if (p1 !== null && typeof p1 === 'number') p1 = Math.round(p1 * 100) / 100;
      if (p2 !== null && typeof p2 === 'number') p2 = Math.round(p2 * 100) / 100;
      
      var nextLevel = level + 1;
      var cost = (nextLevel <= 10) ? getCost(card.rarity, nextLevel) : {cards:0, coins:0};
      
      outputRows.push([
        card.id,
        card.name,
        card.rarity,
        card.type,
        card.param1_name,
        card.param2_name,
        level,
        p1,
        p2,
        cost.cards,
        cost.coins
      ]);
    }

    sheet.getRange(2, 1, outputRows.length, headers.length).setValues(outputRows);
  }
