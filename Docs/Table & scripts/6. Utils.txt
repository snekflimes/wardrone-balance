var Formatting = {};

Formatting.setupFormatting = function(spreadsheet) {
  var sheets = spreadsheet.getSheets();
  sheets.forEach(function(sheet) {
    if (sheet.getLastRow() > 0) {
      var headers = sheet.getRange(1, 1, 1, sheet.getLastColumn());
      headers.setBackground('#1a73e8').setFontWeight('bold').setFontColor('#FFFFFF');
      
      var secondRow = sheet.getRange(2, 1, 1, sheet.getLastColumn());
      if (sheet.getRange(2, 1).getValue() !== '') {
        secondRow.setBackground('#e8f0fe').setFontWeight('bold');
      }
      
      if (sheet.getName() === 'Constants') {
        sheet.setColumnWidth(1, 300);
        sheet.setColumnWidth(2, 120);
        sheet.setColumnWidth(3, 400);
      }
    }
    sheet.autoResizeColumns(1, sheet.getLastColumn());
  });
};

DataRefresh.refreshAllData = function() {
  var spreadsheet = SpreadsheetApp.getActiveSpreadsheet();
  
  try {
    spreadsheet.toast('Начинаю обновление данных...', 'Обновление', 3);
    
    this.fillConstants();
    
    this.refreshWeaponsFormulas();
    this.refreshEnemiesFormulas();
    this.refreshWavesFormulas();
    this.refreshCombatSimulatorFormulas();
    this.refreshProgressionFormulas();
    this.refreshCardsFormulas();
    this.refreshMissionsFormulas();
    this.refreshQuestsFormulas();
    this.refreshStatisticsFormulas();
    this.refreshChestsFormulas();
    this.refreshShopFormulas();
    this.refreshBattlePassFormulas();
    this.refreshWeaponModifiersFormulas();
    
    spreadsheet.toast('Все данные успешно обновлены!', 'Готово', 5);

  } catch (error) {
    SpreadsheetApp.getUi().alert('❌ Ошибка при обновлении: ' + error.toString());
  }
};

DataRefresh.refreshWeaponsFormulas = function() {
  var weaponsSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Weapons');
  if (!weaponsSheet) return;
  
  var maxLevel = 10;
  
  for (var level = 1; level <= maxLevel; level++) {
    var row = level + 1;
    
    weaponsSheet.getRange('C' + row).setFormula('=Constants!$B$3*POWER(Constants!$B$6,' + (level-1) + ')');
    weaponsSheet.getRange('D' + row).setFormula('=Constants!$B$21*POWER(Constants!$B$22,' + (level-1) + ')');
    weaponsSheet.getRange('E' + row).setFormula('=C' + row + '*D' + row + '/60');
    weaponsSheet.getRange('F' + row).setFormula('=Constants!$B$27*POWER(Constants!$B$28,' + (level-1) + ')');
    weaponsSheet.getRange('G' + row).setFormula('=Constants!$B$7*300*POWER(Constants!$B$7,' + (level-1) + ')');
  }
  
  var rocketStartRow = maxLevel + 3;
  for (var level = 1; level <= maxLevel; level++) {
    var row = rocketStartRow + level - 1;
    
    weaponsSheet.getRange('C' + row).setFormula('=Constants!$B$4*POWER(Constants!$B$6,' + (level-1) + ')');
    weaponsSheet.getRange('D' + row).setFormula('=Constants!$B$23*POWER(Constants!$B$24,' + (level-1) + ')');
    weaponsSheet.getRange('E' + row).setFormula('=C' + row + '*D' + row + '/60');
    weaponsSheet.getRange('F' + row).setFormula('=Constants!$B$29*POWER(Constants!$B$30,' + (level-1) + ')');
    weaponsSheet.getRange('G' + row).setFormula('=Constants!$B$7*500*POWER(Constants!$B$7,' + (level-1) + ')');
  }
  
  var guidedStartRow = maxLevel * 2 + 4;
  for (var level = 1; level <= maxLevel; level++) {
    var row = guidedStartRow + level - 1;
    
    weaponsSheet.getRange('C' + row).setFormula('=Constants!$B$5*POWER(Constants!$B$6,' + (level-1) + ')');
    weaponsSheet.getRange('D' + row).setFormula('=Constants!$B$25*POWER(Constants!$B$26,' + (level-1) + ')');
    weaponsSheet.getRange('E' + row).setFormula('=C' + row + '*D' + row + '/60');
    weaponsSheet.getRange('F' + row).setFormula('=Constants!$B$31*POWER(Constants!$B$32,' + (level-1) + ')');
    weaponsSheet.getRange('G' + row).setFormula('=Constants!$B$7*800*POWER(Constants!$B$7,' + (level-1) + ')');
  }
};
